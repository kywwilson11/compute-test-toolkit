% ---- Guide PDF styling (pandoc + xelatex, default Latin Modern fonts) ----
% We deliberately use the default LaTeX fonts: Latin Modern Math covers every math
% symbol, so we route Unicode symbols in prose to math mode via newunicodechar and
% they always render (no system-font coverage gaps).

% Wrap long lines in highlighted (language-tagged) code blocks.
\usepackage{fvextra}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{breaklines,breakanywhere,commandchars=\\\{\},fontsize=\small}

% Shrink plain verbatim (ASCII diagrams) slightly so wide ones fit.
\usepackage{etoolbox}
\AtBeginEnvironment{verbatim}{\small}

\usepackage{microtype}

% Widen the TOC number boxes so multi-digit chapter/section numbers (e.g. "13.10")
% never run into the entry title. Class-safe: \cftchapnumwidth only exists under
% report/book, so guard it (the per-guide builds use the article class).
\usepackage{tocloft}
\setlength{\cftsecnumwidth}{3.2em}
\setlength{\cftsubsecnumwidth}{4em}
\makeatletter
\@ifundefined{cftchapnumwidth}{}{\setlength{\cftchapnumwidth}{2.6em}}
\makeatother
